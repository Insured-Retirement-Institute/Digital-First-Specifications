openapi: "3.1.0"
info:
  title: Document Processing API
  version: "1.0.0"
  description: >
    Standardized API for secure exchange of unstructured data (imaged documents, signatures, forms)
    during pre-sale, new business, and post-issue processing of annuity and life insurance transactions.
    This API follows an asynchronous model:
    1. Sender submits attachments -> receives immediate 202 acknowledgment.
    2. DTCC forwards to Data Provider -> Data Provider responds with success/failure.
    3. DTCC sends final confirmation notification back to the original sender.

    description: Submit attachment documents for async processing
  - name: Notifications
    description: Final confirmation notifications sent to the original sender
paths:
  /v1/attachments:
    post:
      operationId: submitAttachments
      tags:
        - Attachments
      summary: Submit forms detail and one file per form
      description: >-
        Asynchronous multipart request with a JSON payload part (attachmentRequestBody) and one or more
        binary file parts in 'attachmentRequestFiles[]'.
        Each 'forms[i]' must include an 'attachment.fileName' that matches exactly one uploaded file.
        Combined size of all files must be <= 30MB.
        Returns an immediate 202 acknowledgment. Final processing outcome is delivered via the
        /v1/attachments/notifications endpoint.
      parameters:
        - $ref: '#/components/parameters/correlationId'
        - $ref: '#/components/parameters/dtcc-trans-ref-guid'
        - $ref: '#/components/parameters/dtcc-sender'
        - $ref: '#/components/parameters/dtcc-receiver'
        - $ref: '#/components/parameters/dtcc-message-type'
        - $ref: '#/components/parameters/dtcc-sub-message-type'
        - $ref: '#/components/parameters/dtcc-url-params'
        - $ref: '#/components/parameters/dtcc-api-version'
      requestBody:
        content:
          multipart/form-data:
            schema:
              type: object
              required:
                - attachmentRequestBody
                - attachmentRequestFiles
              properties:
                attachmentRequestBody:
                  $ref: "#/components/schemas/attachmentRequestBody"
                attachmentRequestFiles:
                  type: array
                  description: The attachmentRequestFiles Resource is designed to add the multipart files.
                  minItems: 1
                  maxItems: 20
                  items:
                    type: string
                    format: binary
              encoding:
                attachmentRequestBody:
                  contentType: application/json
      responses:
        '202':
          description: Request accepted for processing.
          headers:
            CorrelationId:
              $ref: '#/components/headers/CorrelationId'
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/acknowledgmentResponse"
              example:
                dtcc-trans-ref-guid: "ABCD-2025-00001"
                messageStatus:
                  statusCode: "202"
                  description: "Accepted for processing"
        '400': { $ref: '#/components/responses/BadRequestError' }
        '401': { $ref: '#/components/responses/UnauthorizedError' }
        '403': { $ref: '#/components/responses/ForbiddenError' }
        '404': { $ref: '#/components/responses/NotFoundError' }
        '405': { $ref: '#/components/responses/MethodNotAllowedError' }
        '406': { $ref: '#/components/responses/NotAcceptableError' }
        '413': { $ref: '#/components/responses/PayloadTooLargeError' }
        '415': { $ref: '#/components/responses/UnsupportedMediaTypeError' }
        '429': { $ref: '#/components/responses/TooManyRequestsError' }

  /v1/attachments/notifications:

      operationId: sendAttachmentNotification
      tags:

      summary: Final confirmation notification to the original sender
      description: >-
        Delivers the final processing outcome (success or failure) from the Data Provider back to the
        original sender. This endpoint is called by DTCC after receiving the Data Provider's response.
        The originalCorrelationId in the response body allows the sender to correlate this notification
        to their original submission.
      parameters:
        - $ref: '#/components/parameters/correlationId'

        - $ref: '#/components/parameters/dtcc-sender'

        - $ref: '#/components/parameters/dtcc-sub-message-type'
        - $ref: '#/components/parameters/dtcc-url-params'
        - $ref: '#/components/parameters/dtcc-api-version'
      requestBody:
        required: true
        content:
          application/json:
            schema:
